!
ipv6 dhcp pool DEPT3_STATEFUL
 address prefix 2001:DB8:ACAD:30::/64 lifetime 172800 86400

 domain-name propiedadhorizontal.local

interface GigabitEthernet0/0.30
 description Gateway_Departamento_3_DHCPv6_Stateful
 encapsulation dot1Q 30

 ipv6 address 2001:DB8:ACAD:30::1/64
 ipv6 enable
 ipv6 nd managed-config-flag
 ipv6 dhcp server DEPT3_STATEFUL
!
