! =====================================================================
! Fragmento de referencia: Configuración DHCPv6 Stateless en R1
! Aplica a: Departamento 2 (VLAN 20) y Departamento 4 (VLAN 40)
! Mecanismo: SLAAC genera la GUA (/64) + DHCPv6 entrega DNS y Dominio (O=1, M=0)
! =====================================================================
ipv6 unicast-routing
!
ipv6 dhcp pool DEPT2_STATELESS
 dns-server 2001:DB8:ACAD:99::10
 domain-name propiedadhorizontal.local
!
ipv6 dhcp pool DEPT4_STATELESS
 dns-server 2001:DB8:ACAD:99::10
 domain-name propiedadhorizontal.local
!
interface GigabitEthernet0/0.20
 description Gateway_Departamento_2_DHCPv6_Stateless
 encapsulation dot1Q 20
 ipv6 address FE80::1 link-local
 ipv6 address 2001:DB8:ACAD:20::1/64
 ipv6 enable
 ipv6 nd other-config-flag
 ipv6 dhcp server DEPT2_STATELESS
!
interface GigabitEthernet0/0.40
 description Gateway_Departamento_4_DHCPv6_Stateless
 encapsulation dot1Q 40
 ipv6 address FE80::1 link-local
 ipv6 address 2001:DB8:ACAD:40::1/64
 ipv6 enable
 ipv6 nd other-config-flag
 ipv6 dhcp server DEPT4_STATELESS
!
